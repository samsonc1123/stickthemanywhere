<script>
const boltsCanvas = document.getElementById('bolts');
const noiseCanvas = document.getElementById('noise');
const flash = document.getElementById('flash');
const bctx = boltsCanvas.getContext('2d');
const nctx = noiseCanvas.getContext('2d');

function resize(){
  [boltsCanvas, noiseCanvas].forEach(c=>{
    c.width = window.innerWidth;
    c.height = window.innerHeight;
  });
}
resize();
window.addEventListener('resize', resize);

// ---- static noise layer ----
function drawNoise(){
  const w = noiseCanvas.width, h = noiseCanvas.height;
  const imgData = nctx.createImageData(w,h);
  for(let i=0;i<imgData.data.length;i+=4){
    const v = Math.random()*255;
    imgData.data[i]=v; imgData.data[i+1]=v; imgData.data[i+2]=v; imgData.data[i+3]=255;
  }
  nctx.putImageData(imgData,0,0);
}
setInterval(drawNoise, 100);

// ---- procedural lightning bolt (midpoint displacement) ----
function generateBolt(x1,y1,x2,y2,displace,segments,branchChance){
  // returns array of {points:[[x,y],...]} main + branches
  const branches = [];
  function subdivide(x1,y1,x2,y2,disp,depth){
    if (depth<=0 || disp<2){
      return [[x1,y1],[x2,y2]];
    }
    const mx = (x1+x2)/2 + (Math.random()-0.5)*disp;
    const my = (y1+y2)/2 + (Math.random()-0.5)*disp;
    if (Math.random() < branchChance && depth < segments){
      const bx = mx + (Math.random()-0.5)*disp*3;
      const by = my + (Math.random()-0.5)*disp*3 + disp;
      branches.push(subdivide(mx,my,bx,by, disp*0.5, depth-1));
    }
    const left = subdivide(x1,y1,mx,my, disp*0.55, depth-1);
    const right = subdivide(mx,my,x2,y2, disp*0.55, depth-1);
    return left.concat(right.slice(1));
  }
  const main = subdivide(x1,y1,x2,y2,displace,segments);
  return {main, branches};
}

function drawPath(ctx, pts, width, alpha){
  ctx.beginPath();
  ctx.moveTo(pts[0][0], pts[0][1]);
  for(let i=1;i<pts.length;i++) ctx.lineTo(pts[i][0], pts[i][1]);
  ctx.strokeStyle = `rgba(160,220,255,${alpha})`;
  ctx.lineWidth = width;
  ctx.shadowColor = '#7fd4ff';
  ctx.shadowBlur = 14;
  ctx.stroke();
}

function strikeBolt(x1,y1,x2,y2, opts={}){
  const {segments=6, displace=Math.abs(x2-x1)*0.5+40, branchChance=0.35, duration=180, doFlash=false} = opts;
  const bolt = generateBolt(x1,y1,x2,y2,displace,segments,branchChance);
  let start = null;
  if (doFlash){
    flash.style.transition = 'none';
    flash.style.opacity = 0.12;
    setTimeout(()=>{ flash.style.transition='opacity .3s ease'; flash.style.opacity=0; }, 40);
  }
  function frame(t){
    if(!start) start=t;
    const elapsed = t-start;
    bctx.clearRect(0,0,boltsCanvas.width, boltsCanvas.height);
    const alpha = 1 - elapsed/duration;
    if (alpha > 0){
      // faint outer glow pass
      drawPath(bctx, bolt.main, 4, alpha*0.25);
      drawPath(bctx, bolt.main, 1.6, alpha);
      bolt.branches.forEach(b => drawPath(bctx, b, 1, alpha*0.8));
      requestAnimationFrame(frame);
    } else {
      bctx.clearRect(0,0,boltsCanvas.width, boltsCanvas.height);
    }
  }
  requestAnimationFrame(frame);
}

// ---- ambient random strikes across the page ----
function ambientStrike(){
  const w = boltsCanvas.width, h = boltsCanvas.height;
  const x1 = Math.random()*w;
  const x2 = x1 + (Math.random()-0.5)*160;
  strikeBolt(x1, 0, x2, h*0.35 + Math.random()*h*0.4, {doFlash: Math.random()<0.3});
  const next = 1400 + Math.random()*2600;
  setTimeout(ambientStrike, next);
}
setTimeout(ambientStrike, 900);

// ---- cursor arc: charge builds near edges, arcs jump to cursor ----
let mouseX = null, mouseY = null;
window.addEventListener('mousemove', e=>{
  mouseX = e.clientX; mouseY = e.clientY;
});

function maybeCursorArc(){
  if (mouseX !== null){
    const w = boltsCanvas.width;
    const distFromEdge = Math.min(mouseX, w-mouseX);
    // more likely to arc the closer you are to an edge
    const chance = Math.max(0, 1 - distFromEdge/220);
    if (Math.random() < chance*0.6){
      const fromLeft = mouseX < w/2;
      const sx = fromLeft ? 0 : w;
      const sy = mouseY + (Math.random()-0.5)*80;
      strikeBolt(sx, sy, mouseX, mouseY, {segments:5, displace:50, branchChance:0.25, duration:140});
    }
  }
  setTimeout(maybeCursorArc, 250);
}
maybeCursorArc();
